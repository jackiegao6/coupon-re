cmake_minimum_required(VERSION 3.16)
project(couponim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(coupon_stream streaming_main.cpp)
add_executable(coupon_experiment experiment_main.cpp)
foreach(target coupon_stream coupon_experiment)
    if(MSVC)
        target_compile_options(${target} PRIVATE /O2 /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -O3 -ffp-contract=off)
    endif()
endforeach()
if(WIN32)
    target_link_libraries(coupon_stream PRIVATE psapi)
endif()
